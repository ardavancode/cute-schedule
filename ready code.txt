"use client";

import React, { useState } from "react";
import { motion } from "framer-motion";
import { Search, User, Moon, Sun } from "lucide-react";
import { cn } from "@/lib/utils";

interface ThemeToggleProps {
  className?: string;
}

function ThemeToggle({ className }: ThemeToggleProps) {
  const [isDark, setIsDark] = useState(true);

  return (
    <div
      className={cn(
        "flex w-12 h-6 p-0.5 rounded-full cursor-pointer transition-all duration-300",
        isDark 
          ? "bg-zinc-950 border border-zinc-800" 
          : "bg-white border border-[#FFD93D]/30",
        className
      )}
      onClick={() => setIsDark(!isDark)}
      role="button"
      tabIndex={0}
    >
      <div className="flex justify-between items-center w-full relative">
        <motion.div
          className={cn(
            "flex justify-center items-center w-5 h-5 rounded-full absolute",
            isDark 
              ? "bg-zinc-800" 
              : "bg-[#FFD93D]"
          )}
          animate={{
            x: isDark ? 0 : 20,
          }}
          transition={{ type: "spring", stiffness: 500, damping: 30 }}
        >
          {isDark ? (
            <Moon 
              className="w-3 h-3 text-white" 
              strokeWidth={1.5}
            />
          ) : (
            <Sun 
              className="w-3 h-3 text-[#333333]" 
              strokeWidth={1.5}
            />
          )}
        </motion.div>
      </div>
    </div>
  );
}

interface CuteScheduleHeaderProps {
  appName?: string;
  onSearchClick?: () => void;
  onUserClick?: () => void;
}

function CuteScheduleHeader({ 
  appName = "Cute Schedule",
  onSearchClick = () => {},
  onUserClick = () => {}
}: CuteScheduleHeaderProps) {
  return (
    <motion.header
      initial={{ y: -100, opacity: 0 }}
      animate={{ y: 0, opacity: 1 }}
      transition={{ type: "spring", stiffness: 300, damping: 30 }}
      className="fixed top-4 left-1/2 -translate-x-1/2 z-50 w-full max-w-md px-4"
    >
      <div className="relative flex items-center justify-between gap-4 bg-[#FFFFFF]/90 dark:bg-black/80 backdrop-blur-xl rounded-2xl px-6 py-3 shadow-xl border border-[#FFD93D]/30 dark:border-gray-700/50">
        
        {/* Floating glow effect */}
        <div className="absolute inset-0 bg-gradient-to-r from-[#FF6B6B]/20 via-[#FFD93D]/20 to-[#FF6B6B]/10 rounded-2xl blur-xl -z-10" />
        
        {/* App Name */}
        <motion.h1 
          className="text-lg font-bold bg-gradient-to-r from-[#FF6B6B] via-[#FFD93D] to-[#FF6B6B] bg-clip-text text-transparent"
          whileHover={{ scale: 1.05 }}
          transition={{ type: "spring", stiffness: 400, damping: 20 }}
        >
          {appName}
        </motion.h1>

        {/* Right side controls */}
        <div className="flex items-center gap-3">
          {/* Search Icon */}
          <motion.button
            onClick={onSearchClick}
            whileHover={{ scale: 1.1 }}
            whileTap={{ scale: 0.95 }}
            className="flex items-center justify-center w-8 h-8 text-[#666666] dark:text-gray-300 hover:text-[#FF6B6B] dark:hover:text-[#FF6B6B] transition-colors rounded-lg hover:bg-[#FFF5E1]/70 dark:hover:bg-gray-800/50"
          >
            <Search size={18} strokeWidth={2} />
          </motion.button>

          {/* Theme Toggle */}
          <ThemeToggle />

          {/* User Icon */}
          <motion.button
            onClick={onUserClick}
            whileHover={{ scale: 1.1 }}
            whileTap={{ scale: 0.95 }}
            className="flex items-center justify-center w-8 h-8 text-[#666666] dark:text-gray-300 hover:text-[#FF6B6B] dark:hover:text-[#FF6B6B] transition-colors rounded-lg hover:bg-[#FFF5E1]/70 dark:hover:bg-gray-800/50"
          >
            <User size={18} strokeWidth={2} />
          </motion.button>
        </div>
      </div>
    </motion.header>
  );
}

      />

          <div className="h-32 bg-[#FFF5E1]/80 dark:bg-black/20 rounded-xl backdrop-blur-sm border border-[#FFD93D]/20 dark:border-gray-700/50" />
          <div className="h-32 bg-[#FFF5E1]/80 dark:bg-black/20 rounded-xl backdrop-blur-sm border border-[#FFD93D]/20 dark:border-gray-700/50" />
          <div className="h-32 bg-[#FFF5E1]/80 dark:bg-black/20 rounded-xl backdrop-blur-sm border border-[#FFD93D]/20 dark:border-gray-700/50" />
